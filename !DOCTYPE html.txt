  <script>
    // الوضع الليلي🌚/النهاري
    function toggleMode() {
      const body = document.body;
      body.classList.toggle("dark-mode");
      body.classList.toggle("light-mode");
    }

    //💻 الآلة الحاسبة
    function append(char) {
      document.getElementById('display').value += char;
    }

    function clearDisplay() {
      document.getElementById('display').value = '';
    }

    function calculate() {
      try {
        document.getElementById('display').value = eval(document.getElementById('display').value);
      } catch {
        document.getElementById('display').value = '❌خطأ';
      }
    }

    // التبديل بين الآلة ولعبة الثعبان
    function toggleView() {
      const calc = document.querySelector(".calculator");
      const snake = document.getElementById("snakeGame");
      const btn = document.getElementById("toggleBtn");

      if (calc.style.display !== "none") {
        calc.style.display = "none";
        snake.style.display = "block";
        btn.innerText = "رجوع💻";
        startGame();
      } else {
        calc.style.display = "block";
        snake.style.display = "none";
        btn.innerText = "تبديل إلى لعبة الثعبان";
      }
    }

    // لعبة الثعبان
    let canvas, ctx, snake, food, dx, dy, gameInterval;

    function startGame() {
      clearInterval(gameInterval);
      canvas = document.getElementById("gameCanvas");
      ctx = canvas.getContext("2d");
      snake = [{x: 150, y: 150}];
      food = spawnFood();
      dx = 10;
      dy = 0;
      document.removeEventListener("keydown", changeDirection);
      document.addEventListener("keydown", changeDirection);
      gameInterval = setInterval(update, 100);
      const oldBtn = document.getElementById("restartBtn");
      if (oldBtn) oldBtn.remove();
    }

    function update() {
      const head = {x: snake[0].x + dx, y: snake[0].y + dy};
      snake.unshift(head);
      if (head.x === food.x && head.y === food.y) {
        food = spawnFood();
      } else {
        snake.pop();
      }

      if (head.x < 0 || head.x >= 300 || head.y < 0 || head.y >= 300 || checkCollision()) {
        clearInterval(gameInterval);
        showGameOver();
        return;
      }

      ctx.fillStyle = "black";
      ctx.fillRect(0, 0, 300, 300);
      ctx.fillStyle = "lime";
      snake.forEach(part => ctx.fillRect(part.x, part.y, 10, 10));
      ctx.fillStyle = "red";
      ctx.fillRect(food.x, food.y, 10, 10);
    }

    function changeDirection(e) {
      if (e.key === "ArrowUp" && dy === 0) setDirection('up');
      else if (e.key === "ArrowDown" && dy === 0) setDirection('down');
      else if (e.key === "ArrowLeft" && dx === 0) setDirection('left');
      else if (e.key === "ArrowRight" && dx === 0) setDirection('right');
    }

    function setDirection(dir) {
      if (dir === 'up' && dy === 0) { dx = 0; dy = -10; }
      else if (dir === 'down' && dy === 0) { dx = 0; dy = 10; }
      else if (dir === 'left' && dx === 0) { dx = -10; dy = 0; }
      else if (dir === 'right' && dx === 0) { dx = 10; dy = 0; }
    }

    function spawnFood() {
      return {
        x: Math.floor(Math.random() * 30) * 10,
        y: Math.floor(Math.random() * 30) * 10
      };
    }

    function checkCollision() {
      for (let i = 1; i < snake.length; i++) {
        if (snake[i].x === snake[0].x && snake[i].y === snake[0].y) {
          return true;
        }
      }
      return false;
    }

    function showGameOver() {
      const controls = document.getElementById("controls");

      // تحقق: هل الزر موجود؟
      if (document.getElementById("restartBtn")) return;

      const restartBtn = document.createElement("button");
      restartBtn.innerText = "🔴إعادة المحاولة";
      restartBtn.id = "restartBtn";
      restartBtn.className = "btn-game";
      restartBtn.onclick = () => {
        restartBtn.remove();
        startGame();
      };

      controls.appendChild(restartBtn);
    }
  </script>
